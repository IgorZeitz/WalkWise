import pickle
import numpy as np
from sklearn.model_selection import train_test_split
from sklearn.decomposition import PCA
from sklearn.svm import SVC
from sklearn.pipeline import make_pipeline
from sklearn.preprocessing import StandardScaler
from sklearn.metrics import classification_report
import matplotlib.pyplot as plt
import cv2

# === 1. Wczytanie danych ===
with open("X_9_RGB.pickle", "rb") as f:
    X = pickle.load(f)

with open("y_9_RGB.pickle", "rb") as f:
    y = pickle.load(f)





# === 2. Przekształcenie danych ===
X = np.array(X)  # Lista → numpy
X_resized = np.array([cv2.resize(img, (16,16), interpolation=cv2.INTER_AREA) for img in X])
n_samples = len(X_resized)
X_flat = X_resized.reshape((n_samples, -1))  # teraz 16*16*3 = 768 cech

print(f"Liczba próbek: {n_samples}")
print(f"Rozmiar obrazu: {X.shape[1:]} → Wektor długości {X_flat.shape[1]}")

# === 3. Podział na zbiór treningowy i testowy ===
X_train, X_test, y_train, y_test = train_test_split(X_flat, y, test_size=0.2, random_state=42)

# === 4. Pipeline: Skalowanie → PCA → SVM ===
model = make_pipeline(
    StandardScaler(),
    PCA(n_components=100),  # Możesz zmniejszyć lub zwiększyć
    SVC(kernel="rbf", C=10, gamma="scale")
)

# === 5. Trening ===
print("Trwa trenowanie modelu...")
model.fit(X_train, y_train)
print("Gotowe!")

# === 6. Test ===
y_pred = model.predict(X_test)
print("\nRaport klasyfikacji:")
print(classification_report(y_test, y_pred))
